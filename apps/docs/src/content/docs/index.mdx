## Quick Start

```bash
# Initialize in your project
npx @thaparoyal/calendar-cli init

# Add the calendar component
npx @thaparoyal/calendar-cli add calendar
```

## Example

```tsx
import { Calendar } from '@thaparoyal/calendar-react';

function App() {
  const [date, setDate] = useState(null);

  return (
    <Calendar.Root
      config={{ calendarType: 'BS', locale: 'en' }}
      value={date}
      onValueChange={setDate}
    >
      <Calendar.Header>
        <Calendar.PrevButton />
        <Calendar.Title />
        <Calendar.NextButton />
      </Calendar.Header>
      <Calendar.Grid>
        <Calendar.GridHead />
        <Calendar.GridBody />
      </Calendar.Grid>
    </Calendar.Root>
  );
}
```
